cmake_minimum_required(VERSION 3.16)
project(server_lock_order LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(lock_order_regression lock_order_regression.cpp)
target_compile_features(lock_order_regression PRIVATE cxx_std_17)
target_compile_options(lock_order_regression PRIVATE -Wall -Wextra -Werror)
target_link_libraries(lock_order_regression PRIVATE Threads::Threads)
enable_testing()
add_test(NAME lock_order_regression COMMAND lock_order_regression)
set_tests_properties(lock_order_regression PROPERTIES TIMEOUT 20)
